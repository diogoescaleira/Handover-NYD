  WHEN zx.contraparte_i_desc = "outras empresas nao financeiras" AND flag_pme = "1" THEN "c002x   - Micro & SME"

  WHEN zx.contraparte_i_desc = "particulares" THEN "c001x   - Households"

"Rank 9 - Ranking in insolvency (master scale)" as Insolvency_ranking, -- Confirmar se este campo é estatico

IF(zx.cod_ref_cq="" OR zx.cod_ref_cq IS NULL, zx.zcliente1, zx.cod_ref_cq) as Master_agreement_ID,

"" Type_of_MA, -- Vem sempre a vazio e não é suposto

IF(zx.cod_tip_contrato = "ISDA", "Yes - ISDA Universal Protocol", "Yes - Other Agreement for resolution stay recognition") Resolution_Stay_Recognition,

zx.gcliente as Entity_name_of_counterparty,

IF(zx.clei = "#" OR zx.clei = "" OR zx.clei IS NULL, zx.zcliente1, zx.clei) as Counterparty_identifier,

IF(zx.clei = "#" OR zx.clei = "" OR zx.clei IS NULL, "Type of identifier, other than LEI or MFI code", "LEI code") as Type_of_identifier,

NVL(zx.tayd91c0_gelem30, "0") as Country_of_counterparty,

IF(zx.cod_tip_contrato = "ISDA", "UNITED KINGDOM", "PORTUGAL") as Governing_law_MA_single_deal, -- Tenho duvidas que este campo seja apenas isto

zx.Calculation as Net_MTM_Value,

IF(zx.Calculation2 IS NULL, 0, -zx.Calculation2)  as Value_of_net_collateral,

FROM bu_captools_work.QUERY_FOR_AL015_ACT_FIN_0001 as zx;
