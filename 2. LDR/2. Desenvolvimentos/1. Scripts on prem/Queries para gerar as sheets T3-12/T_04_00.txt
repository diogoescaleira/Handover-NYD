CREATE TABLE bu_captools_work.T_04_00 AS
SELECT

cast(row_number() over (order by zx.depositario1) as string) as Number,

CASE
  WHEN zx.tipo_titulo1 = "Obrigacao Subordinada" THEN "r0531 - o/w (part of) subordinated liabilities recognised as own funds"
  WHEN zx.tipo_titulo1 = "Obrigacao Hipotecaria" THEN "r0120 - Secured liabilities - collateralized part (BRRD art. 44/2/b)"
  WHEN zx.tipo_titulo1 = "Titularizacao Sintetica" OR zx.nome_atributo_dim5 = "Obrigações estruturadas" THEN "r0350 - Structured notes"
  ELSE ""
END as Line,

IF(zx.tipo_titulo1 = "Obrigacao Subordinada", "c001x   - Households", "c004x   - Institutions") as Column_T_01,

CASE
  WHEN zx.tipo_titulo1 = "Obrigacao Subordinada" THEN "Rank 3 - Ranking in insolvency (master scale)"
  WHEN zx.tipo_titulo1 = "Obrigacao Hipotecaria" THEN "Rank 20 - Ranking in insolvency (master scale)"
  WHEN zx.tipo_titulo1 = "Titularizacao Sintetica" OR zx.nome_atributo_dim5 = "Obrigações estruturadas" THEN "Rank 9 - Ranking in insolvency (master scale)"
  ELSE ""
END as Insolvency_ranking,

zx.isin1 as ISIN,

CASE
  WHEN zx.tipo_titulo1 = "Obrigacao Subordinada" THEN "Other type of securities"
  WHEN zx.tipo_titulo1 = "Obrigacao Hipotecaria" THEN "Covered Bond"
  WHEN zx.tipo_titulo1 = "Titularizacao Sintetica" OR zx.nome_atributo_dim5 = "Obrigações estruturadas" THEN "Structured notes"
  ELSE ""
END as Type_of_instrument,

zx.tayd91c0_gelemtab1 as Governing_law,

"Not applicable (Contractual recognition of bail-in powers)" as If_third_country_law, --Confirmar se este campo é estático

zx.tayd91c0_gelem30 as Currency,

zx.nom_tot_inic as Original_amount_issued,

zx.sum_of_msaldo_final as Outstanding_principal_amount,

zx.SUM_of_juros as Accrued_interest,

IF(zx.tipo_tx1 IN ("VARIAVEL","VARIÁVEL"), "Variable coupon", IF(zx.tipo_tx1 = "FIXA", "Fixed coupon", "N/A")) as Coupon_type,

zx.tx_fixa1 as Current_Coupon,

CONCAT(SUBSTR(zx.dt_emissao1,7,4),"-",SUBSTR(zx.dt_emissao1,4,2),"-",SUBSTR(zx.dt_emissao1,1,2)) as Issuance_date,

CONCAT(SUBSTR(zx.dt_vencim1,7,4),"-",SUBSTR(zx.dt_vencim1,4,2),"-",SUBSTR(zx.dt_vencim1,1,2)) as Earliest_redemption_date,

CONCAT(SUBSTR(zx.dt_vencim1,7,4),"-",SUBSTR(zx.dt_vencim1,4,2),"-",SUBSTR(zx.dt_vencim1,1,2)) as Legal_maturity,

"Private placement" as Public_private_placement, -- Confirmar se este campo é realmente estático

IF(zx.agente_pagador1 = "" OR zx.agente_pagador1 IS NULL, "0", zx.agente_pagador1) as Paying_agent,

IF(zx.trustee1 = "" OR zx.trustee1 IS NULL, "0", zx.trustee1) as Trustee,

zx.depositario1 as Creditor, -- Não sei de onde vem este campo, penso que devia vir do campo depositario1 da query mas não bate certo

"LEI code" as Type_of_identifier, -- De certeza que este campo não é estático, deve ser dependente do Paying agent ou do Creditor

IF(zx.bolsa_valores1 = "" OR zx.bolsa_valores1 IS NULL, "0", zx.bolsa_valores1) as Exchanges_securities_are_listed_on,

0 as Settlement_systems, -- Confirmar se é sempre 0

IF(zx.bolsa_valores1 = "" OR zx.bolsa_valores1 IS NULL, "0", zx.bolsa_valores1) as Register_holder,

IF(zx.bolsa_valores1 = "" OR zx.bolsa_valores1 IS NULL, "0", zx.bolsa_valores1) as Central_securities_depository, -- Estranho haver três campos iguais

IF(zx.tipo_titulo1 = "Obrigacao Hipotecaria", "Secured", "Unsecured") as Secured_unsecured,

zx.nom_tot_inic as Amount_of_pledge_lien_collateral,

0 as Guarantor_if_applicable, -- Confirmar se este campo é sempre 0

0 as Funding_amount, -- Confirmar se este campo é sempre 0

IF(zx.termo_estruturado1 = "Estruturada", "Structured", IF(zx.termo_estruturado1 = "Nao estruturada","Non-structured/Vanilla","")) as Structured_or_other_non_standard,

"" as Amount_meeting_MREL, -- Confirmar se este campo deve vir sempre a vazio

IF(ZX.eleg_fund_prop1 = "GF2","Grandfathered T2","No") as Qualifying_as_own_funds,

IF(CAST(zx.mon_fund_prop1 as STRING)="" OR zx.mon_fund_prop1 IS NULL, 0, zx.mon_fund_prop1) as Amount_in_Own_Funds

FROM bu_captools_work.QUERY_FOR_FILTER_FOR_QUERY__0027 as zx;
