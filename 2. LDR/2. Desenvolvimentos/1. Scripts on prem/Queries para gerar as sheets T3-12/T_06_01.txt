CREATE TABLE bu_captools_work.T_06_01 AS
SELECT

cast(row_number() over (order by zx.zcliente) as string) as Number,

CASE
  WHEN (zx.classification = "" OR zx.classification IS NULL) AND zx.calculation = 1 THEN "r0180 - Critical service liabilities (BRRD art. 44/2/g/ii)"
  WHEN (zx.classification = "" OR zx.classification IS NULL) AND zx.calculation = 2 THEN "r0190 - Tax and social security authorities liabilities, if preferred (BRRD art. 44/2/g/iii)"
  ELSE zx.classification
END as Line,

CASE
 WHEN zx.contraparte_i_desc = "particulares" OR zx.calculation = 1 THEN "c001x   - Households"
 WHEN zx.contraparte_i_desc = "outras empresas nao financeiras" THEN "c003x   - Corporates"
 WHEN zx.contraparte_i_desc = "instituicoes de credito" THEN "c004x   - Institutions"
 WHEN zx.contraparte_i_desc = "outras instituicoes financeiras" THEN "c005x   - Other financial corporations" -- Condição repetida tem de se alterar ou complementar com mais informação
 WHEN zx.contraparte_i_desc = "setor publico" OR zx.calculation = 2 THEN "c008x   - Government, central banks & supranationals"
 ELSE ""
END as Column_T_01,

IF(zx.contraparte_i_desc = "setor publico" OR zx.calculation = 2, "Rank 16 - Ranking in insolvency (master scale)", "Rank 9 - Ranking in insolvency (master scale)") as Insolvency_ranking,

CASE
 WHEN zx.contraparte_i_desc = "instituicoes de credito" THEN zx.gcliente
 WHEN zx.contraparte_i_desc = "particulares" OR zx.calculation = 1 THEN CONCAT("CSL",array('JAN','FEV','MAR','ABR','MAI','JUN','JUL','AGO','SET','OUT','NOV','DEZ')[CAST(DATE_FORMAT(zx.ref_date,'MM') - 1 as INT)],DATE_FORMAT(zx.ref_date,'YY'))
 WHEN zx.contraparte_i_desc = "setor publico" OR zx.calculation = 2 THEN CONCAT("TL",array('JAN','FEV','MAR','ABR','MAI','JUN','JUL','AGO','SET','OUT','NOV','DEZ')[CAST(DATE_FORMAT(zx.ref_date,'MM') - 1 as INT)],DATE_FORMAT(zx.ref_date,'YY'))
ELSE ""
END as Contract_identifier, -- Nota que para os CASES com o CONCAT na componente final eles estavam a reportar no excel os primeiros dois digitos do ano (que não faz muito sentido) eu implemeti de forma a mostrar os dois ultimos ex. "2022" vinha "20" aqui vem a "22"

CASE
 WHEN zx.contraparte_i_desc = "instituicoes de credito" THEN zx.gcliente
 WHEN zx.contraparte_i_desc = "particulares" OR zx.calculation = 1 THEN CONCAT("Critical Service Liability ",DATE_FORMAT(zx.ref_date,'MMMM')," ",DATE_FORMAT(zx.ref_date,'YYYY'))
 WHEN zx.contraparte_i_desc = "setor publico" OR zx.calculation = 2 THEN CONCAT("Tax Liability ",DATE_FORMAT(zx.ref_date,'MMMM')," ",DATE_FORMAT(zx.ref_date,'YYYY'))
ELSE ""
END as entity_name_of_counterparty,

CASE
 WHEN (zx.clei = "" OR zx.clei = "#" OR zx.clei IS NULL) AND (zx.contraparte_i_desc = "particulares" OR zx.calculation = 1) THEN "CSL001"
 WHEN (zx.clei = "" OR zx.clei = "#" OR zx.clei IS NULL) AND (zx.contraparte_i_desc = "setor publico" OR zx.calculation = 2) THEN "TL001"
 WHEN (zx.clei = "" OR zx.clei = "#" OR zx.clei IS NULL) AND zx.contraparte_i_desc NOT IN ("particulares","setor publico") AND zx.calculation NOT IN (1,2) THEN zx.zcliente
ELSE zx.clei
END as Counterparty_identifier,

IF(zx.clei NOT IN ("","#") AND zx.clei IS NOT NULL,"LEI code","Type of identifier, other than LEI or MFI code") as Type_of_identifier,

"Portugal" as Governing_law, -- Confirmar se é estático

"Financial liabilities other than debt securities issued, deposits, promissory notes, registered notes, bills of exchange, silent partnership contributions" as Type_of_financial_liabilities, -- Confirmar se é estático

"Not applicable (Contractual recognition of bail-in powers)" as If_third_country_law,

zx.SUM_of_msaldo_final as Outstanding_principal_amount,

zx.SUM_juros as Accrued_interest,

IF(zx.AVG_of_tx_contrato_pas="" OR zx.AVG_of_tx_contrato_pas IS NULL, 0, zx.AVG_of_tx_contrato_pas) as current_interest_rate,

zx.tayd91c0_gelem30 as Currency,

zx.ref_date as Issuance_date,

IF(from_unixtime(unix_timestamp(zx.ref_date ,'yyyy-MM-dd'),'u')=5, date_add(zx.ref_date,3), IF(from_unixtime(unix_timestamp(zx.ref_date,'yyyy-MM-dd'),'u')=6, date_add(zx.ref_date,2), date_add(zx.ref_date,1))) AS Earliest_redemption_date, -- Não me parece que esta métrica deva ser sempre o dia util seguinte à data de report 

IF(from_unixtime(unix_timestamp(zx.ref_date ,'yyyy-MM-dd'),'u')=5, date_add(zx.ref_date,3), IF(from_unixtime(unix_timestamp(zx.ref_date,'yyyy-MM-dd'),'u')=6, date_add(zx.ref_date,2), date_add(zx.ref_date,1))) AS Legal_maturity, -- Não me parece que esta métrica deva ser sempre o dia util seguinte à data de report nem que seja sempre igual à Earliest_redemption_date

"Unsecured" as Secured_unsecured, -- Confirmar se é estático

0 as Amount_of_pledge_lien_collateral, -- Confirmar se é estático

0 as Guarantor_if_applicable, -- Confirmar se é estático

0 as Funding_amount, -- Confirmar se é estático

"Non-structured/Vanilla" as Structured_or_other_non_standard, -- Confirmar se é estático

0 as Amount_meeting_MREL, -- Confirmar se é estático

"No" as Qualifying_as_Own_Funds, -- Confirmar se é estático

0 as Amount_Included_in_Own_Funds -- Confirmar se é estático

FROM bu_captools_work.QUERY_FOR_FILTER_FOR_QUERY_0046 as zx;
