

--Tabelas auxiliares para calcular o cpost de 4 digitios com maior contagem para cada cfreguesia 
drop table bu_captools_work.cp7_lau_nuts_aux;
create table  bu_captools_work.cp7_lau_nuts_aux as 
    select lau_code, cp4_code, max(contagem) from (
            select lau_code,cp4_code, count(*) as contagem from (
                    select distinct *,substr(trim(cp7_code),1,4) as cp4_code from bu_captools_work.cp7_lau_nuts
            )aux1  group by 1,2 
    ) aux2 group by 1,2 


drop table bu_captools_work.cp7_lau_nuts_aux2;
create table  bu_captools_work.cp7_lau_nuts_aux2 as
select  a.lau_code, b.cp4_code from (select lau_code,max(_c2) as max_count from bu_captools_work.cp7_lau_nuts_aux group by 1) a
inner join (
            select lau_code, cp4_code, max(contagem) as max_count from (
                select lau_code,cp4_code, count(*) as contagem from (
                    select distinct *,substr(trim(cp7_code),1,4) as cp4_code from bu_captools_work.cp7_lau_nuts
                    )aux1  group by 1,2 
                ) aux2 group by 1,2 
            ) as b
on a.lau_code = b.lau_code  
and a.max_count = b.max_count


select lau_code,count(*) from bu_captools_work.cp7_lau_nuts_aux2 group by 1 having count(*) > 1

select count(*) from (
select aux1.* from bu_captools_work.cp7_lau_nuts_aux2 as aux1
            inner join (select lau_code,count(*) from bu_captools_work.cp7_lau_nuts_aux2 group by 1 having count(*) > 1) aux2
            on aux1.lau_code = aux2.lau_code 
) a
