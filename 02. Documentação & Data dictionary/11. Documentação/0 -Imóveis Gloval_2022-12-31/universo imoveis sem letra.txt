drop table bu_captools_work.ct000;
create table  bu_captools_work.ct000 as

--select count(*)
FROM (select distinct cbalcao, cnumecta, zdeposit, cempresa, zcliente 
            from cd_captools.ct001_univ_saldo 
            where flag_ativo = 1 
            and ref_date = '2022-12-31')  AS ct001

            and ref_date = '2022-12-31') AS ct005

            FROM cd_captools.ct802_codigo_emp 
            WHERE CODIGO_ESPANHA = '00411'
            and ref_date ='2022-12-31') AS ct802
            
on  ct001.cbalcao  = ct005.cbalcao
and ct001.cnumecta = ct005.cnumecta
and ct001.zdeposit = ct005.zdeposit
and ct001.cempresa = ct005.cempresa
and ct001.zcliente = ct005.zcliente
and ct001.cempresa = ct802.CODIGO_ICS
;

on a.ckbalbem = b.ckbalbem

DROP TABLE BU_CAPTOOLS_WORK.imoveis_final_gpt31;
create table BU_CAPTOOLS_WORK.imoveis_final_gpt31 as
select base.*, coalesce(aux.class_gpt31,'') as class_gpt31, coalesce(aux.ind_gpt31,0) as ind_gpt31
from BU_CAPTOOLS_WORK.imoveis_total_dez as base

left join (
        select  a.*, coalesce(b.flag_ienerg,'') AS CLASS_GPT31, b.ind_ienerg AS IND_GPT31
        --select count(*)
        from (select * from BU_CAPTOOLS_WORK.imoveis_total_dez where ind_gloval = 0) as A
        inner join (select distinct ckbalcao, cknumcta, zversao,
                            case
                                when trim(ienerg) in ('A','+') then 'A'
                                when trim(ienerg) in ('B','-') then 'B'
                                when trim(ienerg) in ('C') then 'C'
                                when trim(ienerg) in ('D') then 'D'
                                when trim(ienerg) in ('E') then 'E'
                                when trim(ienerg) in ('F') then 'F'
                                when trim(ienerg) in ('G') then 'G'
                                else ''
                            end as flag_ienerg,
                            1 as ind_ienerg
                   from cd_emprestimos.gpt31_contrato ) as b
        on  a.ckbalcao = b.ckbalcao
        and a.cknumcta = b.cknumcta
        --and cast(a.comp_id_gar as int) = b.zversao
        where flag_ienerg <> ''
    ) as aux
on base.cempresp = aux.cempresp
and base.ckbalres = aux.ckbalres
and base.ckctares = aux.ckctares
and base.ckrefresp = aux.ckrefresp
and base.ckbalcao = aux.ckbalcao
and base.cknumcta = aux.cknumcta
and base.comp_id_gar = aux.comp_id_gar
and base.cempbem = aux.cempbem
and base.ckbalbem = aux.ckbalbem
and base.ckctabem = aux.ckctabem
and base.ckrefbem = aux.ckrefbem
;
