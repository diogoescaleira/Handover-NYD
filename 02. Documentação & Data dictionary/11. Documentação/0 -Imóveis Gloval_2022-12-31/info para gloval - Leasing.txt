drop table bu_captools_work.tabaux_fcl_cpost_cdisconf;
create table bu_captools_work.tabaux_fcl_cpost_cdisconf as
select distinct 
        a.*,
        case
            when cdisconf2 like '%00' and b.cp7_code is not null then b.cp7_code
            when cdisconf2 like '%00' and c.cp4_code is not null then c.cp4_code
            when cdisconf2 like '%0000' and d.cp7_code is not null then d.cp7_code
            when cdisconf2 like '%0000' and f.cp4_code is not null then f.cp4_code
            when length(trim(a.cpost)) = 7  and trim(cpost) <> '0000000' and a.cdisconf = a.cdisconf_gpt18 and upper(trim(a.glocalid)) = upper(trim(a.glocal_gpt18)) then a.cpost
            when length(substr(trim(a.cpost),1,4)) = 4  and substr(trim(a.cpost),1,4) <> '0000' and a.cdisconf = a.cdisconf_gpt18 and upper(trim(a.glocalid)) = upper(trim(a.glocal_gpt18)) then substr(trim(a.cpost),1,4)
            else coalesce(g.cp4_code,'')
        end as codigo_postal
--select count(*)
from bu_captools_work.tabaux_fcl_tatj48 as a

left join (select distinct lau_code,cp7_code from  bu_captools_work.cp7_lau_nuts) as b
on substr(a.cdisconf2,1,4) = substr(trim(b.lau_code),1,4)
and a.cpost = trim(b.cp7_code)

left join (select distinct  lau_code,cp7_code, substr(trim(cp7_code),1,4) as cp4_code from  bu_captools_work.cp7_lau_nuts) as c
on substr(a.cdisconf2,1,4) = substr(trim(c.lau_code),1,4)
and substr(a.cpost,1,4) = c.cp4_code

left join (select distinct lau_code,cp7_code from  bu_captools_work.cp7_lau_nuts) as d
on substr(a.cdisconf2,1,2) = substr(trim(d.lau_code),1,2)
and a.cpost = trim(d.cp7_code)

left join (select distinct  lau_code,cp7_code, substr(trim(cp7_code),1,4) as cp4_code from  bu_captools_work.cp7_lau_nuts) as f
on substr(a.cdisconf2,1,2) = substr(trim(f.lau_code),1,2)
and substr(a.cpost,1,4) = f.cp4_code

left join( select aux1.* from bu_captools_work.cp7_lau_nuts_aux2 as aux1
            inner join (select lau_code,count(*) from bu_captools_work.cp7_lau_nuts_aux2 group by 1 having count(*) = 1) aux2
            on aux1.lau_code = aux2.lau_code  
        ) g
on a.cdisconf2 = trim(g.lau_code)
;
