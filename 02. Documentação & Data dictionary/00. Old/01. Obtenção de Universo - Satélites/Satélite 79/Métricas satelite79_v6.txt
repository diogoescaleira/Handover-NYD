--Criação de tabela de repartição com campos sfcs_categ_n1,sfcs_categ_n2 		
drop table if exists bu_esg_work.rf_metricas_pilar3_ctr_sat;
create table bu_esg_work.rf_metricas_pilar3_ctr_sat as
select  
        a.*,
        case
            when trim(b.sfcs_green_activity_comgloval) like 'Real estate%' then '5 - Real estate'
            when trim(b.sfcs_green_activity_comgloval) like 'Transport%' then '2 - Transport'
            when trim(b.sfcs_green_activity_comgloval) like 'Energy%' then '1 - Energy'
            when c.flag_green_CIB = 1 then c.sfcs_category_I
            else 'N/A'
        end as sfcs_categ_n1,
        case
            when trim(b.sfcs_green_activity_comgloval) =  'Real estate - Building renovation loans' then '5.2 - Energy efficiency equipment in buildings'
            when trim(b.sfcs_green_activity_comgloval) like 'Real estate%' then '5.1 - Construction, refurbishment and purchase of “green” buildings'
            when trim(b.sfcs_green_activity_comgloval) like 'Transport%' then '2.1 - Land Transport'
            when trim(b.sfcs_green_activity_comgloval) like 'Energy%' then '1.1 - Renewable energy production'
            when c.flag_green_CIB = 1 and c.sfcs_category_II = 'Renewable Energy Production' then '1.1 - Renewable energy production'
            else 'N/A'
        end as sfcs_categ_n2
from (select distinct * from bu_esg_work.reparticao_garantias_final) as a
left join   (select distinct 
                    cempresa,
                    cbalcao,
                    cnumecta,
                    zdeposit_ct,
                    sfcs_tag_comgloval,
                    sfcs_green_activity_comgloval,
                    tipo_produto,
                    1 as flag_green
            from bu_esg_work.rf_tradutor_CdG_Master 		--PC:apontar para bu_esg_work
                    where ref_date = "${ref_date}" and sfcs_tag_comgloval = 'Green' and trim(cmetanseg) not like 'EB%') as b
on a.cempresa_ct = b.cempresa
and a.cbalcao_ct = b.cbalcao
and a.cnumecta_ct = b.cnumecta
and a.zdeposit_ct = b.zdeposit_ct

left join (select *, 1 as flag_green_CIB from  bu_captools_work.rf_green_scib_sat84) as c	
on a.cempresa_ct = c.cempresa_ct
and a.cbalcao_ct = c.cbalcao_ct
and a.cnumecta_ct = c.cnumecta_ct
and a.zdeposit_ct = c.zdeposit_ct 
and a.cempresa_fr012 = c.cempresa_fr012
and a.cbalcao_fr012 = c.cbalcao_fr012
and a.cnumecta_fr012 = c.cnumecta_fr012
and a.zdeposit_fr012 = c.zdeposit_fr012 

-- left join (select *, 1 as flag_green_CIB from  bu_esg_work.rf_green_scib) as c		--PC: originalmente estava a tabela rf_green_scib_sat84
-- on a.cnumecta_ct = c.cknumcta															--PC: apontar para bu_esg_work
-- and a.zcliente=c.zcliente
;

                        when substr(nace_code,1,1) = 'B' then concat('NACE','02',right(nace_code,2))
                        when substr(nace_code,1,1) = 'C' then concat('NACE','03',right(nace_code,2))
                        when substr(nace_code,1,1) = 'D' then concat('NACE','04',right(nace_code,2))
                        when substr(nace_code,1,1) = 'E' then concat('NACE','05',right(nace_code,2))
                        when substr(nace_code,1,1) = 'F' then concat('NACE','06',right(nace_code,2))
                        when substr(nace_code,1,1) = 'G' then concat('NACE','07',right(nace_code,2))
                        when substr(nace_code,1,1) = 'H' then concat('NACE','08',right(nace_code,2))
                        when substr(nace_code,1,1) = 'L' then concat('NACE','12',right(nace_code,2))
                        else 'NACE0000'
                    end as nace_esg
            from bu_esg_work.pilar3_physical_riskv2) as prgeral1
on a.`22_counterparty_nuts` = prgeral1.nuts_code
and left(a.nace_esg,8) = prgeral1.nace_esg

-- risco físico proxy para demais operações
left join ( select  *,
                    case
                        when substr(nace_code,1,1) = 'A' then concat('NACE','01',right(nace_code,2))
                        when substr(nace_code,1,1) = 'B' then concat('NACE','02',right(nace_code,2))
                        when substr(nace_code,1,1) = 'C' then concat('NACE','03',right(nace_code,2))
                        when substr(nace_code,1,1) = 'D' then concat('NACE','04',right(nace_code,2))
                        when substr(nace_code,1,1) = 'E' then concat('NACE','05',right(nace_code,2))
                        when substr(nace_code,1,1) = 'F' then concat('NACE','06',right(nace_code,2))
                        when substr(nace_code,1,1) = 'G' then concat('NACE','07',right(nace_code,2))
                        when substr(nace_code,1,1) = 'H' then concat('NACE','08',right(nace_code,2))
                        when substr(nace_code,1,1) = 'L' then concat('NACE','12',right(nace_code,2))
                        else 'NACE0000'
                    end as nace_esg
            from bu_esg_work.pilar3_proxy_physical_risk) as prgeral2
on cast(coalesce(a.cpais_residencia,'620') as int) = prgeral2.iso_code
and left(a.nace_esg,8) = prgeral2.nace_esg
;

---------------------------------------------------------------------------------------------------------------------------------------------------
-- average weighted maturity ----------------------------------------------------------------------------------------------------------------------
---------------------------------------------------------------------------------------------------------------------------------------------------
drop table bu_esg_work.rf_metricas_pilar3_sat79_aux3;
create table bu_esg_work.rf_metricas_pilar3_sat79_aux3 as 
select  a.*,
        case 
            when a.flag_maturity = 0 then null
            else a.amount/b.saldo_satelite
        end as aveg_weight
from bu_esg_work.rf_pilar3_sat79_tabaux2 as a
inner join ( select geo,
                    cnael,
                    15_nace_esg,
                    sociedade_contraparte,
                    cod_ajust,
                    concat(idcomb_satelite,";",excluded_paris,";",env_sust_ccm,";",maturity_esg,";",chronic_changes,";",acute_changes) as id_comb,
                    flag_maturity,
                    sum(amount) as saldo_satelite
            from bu_esg_work.rf_pilar3_sat79_tabaux2
            group by 1,2,3,4,5,6,7 ) as b
on concat(a.idcomb_satelite,";",a.excluded_paris,";",a.env_sust_ccm,";",a.maturity_esg,";",a.chronic_changes,";",a.acute_changes) = b.id_comb
and a.flag_maturity = b.flag_maturity
and a.geo = b.geo
and a.cnael = b.cnael
and a.`15_nace_esg`= b.`15_nace_esg`
and a.sociedade_contraparte = b.sociedade_contraparte
and a.cod_ajust = b.cod_ajust
;


---------------------------------------------------------------------------------------------------------------------------------------------------
-- Tabela final - operações gerais ----------------------------------------------------------------------------------------------------------------
---------------------------------------------------------------------------------------------------------------------------------------------------
drop table bu_esg_work.rf_metricas_pilar3_sat79_aux4;
create table bu_esg_work.rf_metricas_pilar3_sat79_aux4 as
select
    reporting_soc,
    counterparty_soc,
    adjustment_code,
    id_comb,
    amount,
    geo,
    cnae,
    nace,
    case when aveg is null then 0 else aveg end as aveg
from (select 
            '00411' as reporting_soc,
            sociedade_contraparte as counterparty_soc,
            cod_ajust as adjustment_code,
            concat(idcomb_satelite,";",excluded_paris,";",env_sust_ccm,";",maturity_esg,";",chronic_changes,";",acute_changes) as id_comb,
            -round(sum(amount),0) as amount,
            geo,
            cnael as cnae,
            nace_esg as nace,
            ceiling(sum(years_to_maturity * aveg_weight)) as aveg,
            maturity_esg
        
        from bu_esg_work.rf_pilar3_sat79_tabaux3
        group by 1,2,3,4,6,7,8,10
        order by 4,1,2,3,6,7,8 ) a
where amount <> 0
;

-- tabela para atribuição de métricas
drop table bu_esg_work.rf_metricas_pilar3_sat79_adj_aux;
create table bu_esg_work.rf_metricas_pilar3_sat79_adj_aux as
select 
        a.idcomb_satelite, 
        a.sociedade_contraparte,
        a.cod_ajust,
        a.cargabal_ct,
        a.saldo_ct,
        b.cod_imovel,
        b.tipo_adjudicado,
        b.conta_cargabal_capital,
        c.conta_cargabal_imparidade,
        coalesce(b.fiabilidad,c.fiabilidad) as fiabilidad,
        coalesce(b.clase_energetica,c.clase_energetica) as clase_energetica,
        coalesce(b.emisiones,c.emisiones) as emisiones,
        coalesce(b.consumos,c.consumos) as consumos,
        'EXC2' as excluded_paris,
        case
            when b.clase_energetica like 'A%' and b.fiabilidad in ('','1-REAL') then 'ENSM1'
            else 'ENSM2'
        end as env_sust_ccm,
        'MESG4' as maturity_esg,
        case
            when coalesce(prhipotec1.rs_acute,prhipotec2.rs_acute) = 'Yes' then 'ACUT1'
            else 'ACUT2'
        end as acute_changes,
        case
            when coalesce(prhipotec1.rs_chronic,prhipotec2.rs_chronic) = 'Yes' then 'CHRO1'
            else 'CHRO2'
        end as chronic_changes,
        case 
            when cargabal_ct = '' then -saldo_ct
            when b.conta_cargabal_capital in ('1605000','1605010') then b.valor_cargabal
            else -c.imparidade_cargabal
        end as amount
      
from (  select csatelite, idcomb_satelite, sociedade_contraparte, cod_ajust, cargabal_ct, sum(saldo_ct) as saldo_ct 
        from bu_esg_work.rf_pilar3_universo_full where csatelite = 79 and idcomb_satelite like '%MC10%' 
        group by 1,2,3,4,5) as a
        
left join (select * from bu_esg_work.adjudicados_aux_final
            where tipo_adjudicado <> 'Totta URBE' and conta_cargabal_capital in ('1605000','1605010')) as b
on a.cargabal_ct = b.conta_cargabal_capital
        
left join (select * from bu_esg_work.adjudicados_aux_final
            where tipo_adjudicado <> 'Totta URBE' and conta_cargabal_imparidade in ('2642300')) as c
on a.cargabal_ct = c.conta_cargabal_imparidade

-- risco físico real
left join (select * from bu_esg_work.pilar3_physical_riskv2 where nace_code like '%Secured%') as prhipotec1
on coalesce(b.id3_collateral_nuts,c.id3_collateral_nuts) = prhipotec1.nuts_code

-- risco físico proxy
left join (select * from bu_esg_work.pilar3_proxy_physical_risk where nace_code like '%Secured%') as prhipotec2
on prhipotec2.iso_code = 620
;

-- tabela final
drop table bu_esg_work.rf_metricas_pilar3_sat79_adj_aux2;
create table bu_esg_work.rf_metricas_pilar3_sat79_adj_aux2 as
select
    '00411' as reporting_soc,
    sociedade_contraparte as counterparty_soc,
    cod_ajust as adjustment_code,
    concat(idcomb_satelite,";",excluded_paris,";",env_sust_ccm,";",maturity_esg,";",chronic_changes,";",acute_changes) as id_comb,
    sum(round(amount,0)) as amount,
    'GEO3' as geo,
    '' as cnae,
    '' as nace,
    0 as aveg
from bu_esg_work.rf_metricas_pilar3_sat79_adj_aux
group by 1,2,3,4,6,7,8,9
;
