        a.*,

            when a.`13_nace` in ('K65.3.0','K66.1.1','K64.3.0','K66.1.2','K66.3.0')  and a.`31_counterparty_type` = 'outras instituicoes financeiras'             then 'Investment firms'
            when a.`13_nace` in ('K66.1.9','K64.2.0')  and a.`31_counterparty_type` = 'outras instituicoes financeiras'                                           then 'Management firms'
            when a.`13_nace` in ('K65.1.1','K65.1.2','K65.2.0', 'K66.2.1','K66.2.2','K66.2.9')   and a.`31_counterparty_type` = 'outras instituicoes financeiras' then 'Insurance undertakings'
            when a.`31_counterparty_type` = 'outras instituicoes financeiras'                                                                                     then 'Rest of other financial corporation'

        c.flag_green_CIB,
        b.sfcs_green_activity_comgloval,
        b.tipo_produto,
			case
            when trim(b.sfcs_green_activity_comgloval) like 'Real estate%' then '5 - Real estate'
            when trim(b.sfcs_green_activity_comgloval) like 'Transport%'   then '2 - Transport'

            when c.flag_green_CIB = 1                                      then c.sfcs_category_I
            else 'N/A'
			end as 
		sfcs_categ_n1,
			case
            when trim(b.sfcs_green_activity_comgloval) =  'Real estate - Building renovation loans' then '5.2 - Energy efficiency equipment in buildings'
            when trim(b.sfcs_green_activity_comgloval) like 'Real estate%' then '5.1 - Construction, refurbishment and purchase of “green” buildings'
            when trim(b.sfcs_green_activity_comgloval) like 'Transport%' then '2.1 - Land Transport'
            when trim(b.sfcs_green_activity_comgloval) like 'Energy%' then '1.1 - Renewable energy production'
            when c.flag_green_CIB = 1 and c.sfcs_category_II = 'Renewable Energy Production' then '1.1 - Renewable energy production'
            else 'N/A'
			end as 
		sfcs_categ_n2,
			case
            when trim(b.sfcs_green_activity_comgloval) =  'Real estate - Building renovation loans' then 'Building Renovation Loans'
            when trim(b.sfcs_green_activity_comgloval) like 'Real estate%' then 'Other households purpose'
            when trim(b.sfcs_green_activity_comgloval) like 'Transport%' then 'Motor vehicle loans'
            when trim(b.sfcs_green_activity_comgloval) like 'Energy%' then 'Other households purpose'
            when c.sfcs_category_I = 'Energy' then 'Other households purpose'
            else 'N/A'
			end as

            else 'N/A'
			end as

			case

            when trim(c.environmental_objective) = 'Climate change mitigation (CCM)' then 'Climate Change Mitigation'
            when a.flag_colateral = 1 then 'Climate Change Mitigation'
            else 'N/A' 
			end as 
		69_environmental_objective,
        
			case
            when f.geographical_criteria = 'Y' and f.size_criteria = 'Y' and (f.pub_interest_criteria_securities = 'Y' or f.pub_interest_criteria_credit_inst = 'Y'
            or f.pub_interest_criteria_insurance_und = 'Y' or f.pub_interest_criteria_state = 'Y')
            then 'Subject to NFRD'
            else 'Not Subject to NFRD'
			end as 
		flag_nfrd
    
from (select distinct * from  Tabela da repartição das garantias ) as a  

left join   

	(select distinct 
					cempresa,

					cnumecta,

as b
					on a.cempresa_ct = b.cempresa
					and a.cbalcao_ct = b.cbalcao
					and a.cnumecta_ct = b.cnumecta
					and a.zdeposit_ct = b.zdeposit_ct

	(select 		*, 
					1 as flag_green_CIB

					on a.cempresa_ct = c.cempresa_ct
					and a.cbalcao_ct = c.cbalcao_ct
					and a.cnumecta_ct = c.cnumecta_ct
					and a.zdeposit_ct = c.zdeposit_ct

(select  
        a.*,
        b.flag_comp_scib,

			case 
            when c.turnoverelegccm is not null then c.turnoverelegccm
            else b.turnoverelegccm
			end as 
		turnover_eleg_ccm,
			case 
            when c.capexelegccm is not null then c.capexelegccm
            else b.capexelegccm
			end as

			from ( select distinct zcliente  from bu_esg_work.rf_metricas_pilar3_sat79_aux1

(select distinct *, 
				1 as flag_comp_scib 
				from bu_captools_work.pilar3_eligibility_ratios_cib_final) 
as b
				on a.zcliente = b.zcliente 
    
left join 

(select distinct *, 
				1 as flag_comp_pt 
				from bu_captools_work.pilar3_eligibility_ratios_PT_) 
as c
				on a.zcliente = c.zcliente 
) as racios
				on a.zcliente = racios.zcliente

drop table bu_esg_work.rf_pilar3_sat79_aux5;
create table bu_esg_work.rf_pilar3_sat79_aux5 as 
select  a.*,
        case 
            when a.flag_maturity = 0 then null
            else a.amount/b.saldo_satelite
        end as aveg_weight
from bu_captools_work.rf_pilar3_sat79_tabaux2 as a
inner join ( select geo,
                    cnael,
                    nace_esg,
                    sociedade_contraparte,
                    cod_ajust,
                    concat(idcomb_satelite,";",excluded_paris,";",env_sust_ccm,";",maturity_esg,";",chronic_changes,";",acute_changes) as id_comb,
                    flag_maturity,
                    sum(amount) as saldo_satelite
            from bu_esg_work.rf_pilar3_sat79_aux4
            group by 1,2,3,4,5,6,7 ) as b
on concat(a.idcomb_satelite,";",a.excluded_paris,";",a.env_sust_ccm,";",a.maturity_esg,";",a.chronic_changes,";",a.acute_changes) = b.id_comb
and a.flag_maturity = b.flag_maturity
and a.geo = b.geo
and a.cnael = b.cnael
and a.nace_esg = b.nace_esg
and a.sociedade_contraparte = b.sociedade_contraparte
and a.cod_ajust = b.cod_ajust
;

drop table bu_esg_work.rf_pilar3_sat79_aux6;----------Tabela final sem adjudicados
create table bu_esg_work.rf_pilar3_sat79_aux6 as
select
    reporting_soc,
    counterparty_soc,
    adjustment_code,
    id_comb,
    amount,
    geo,
    cnae,
    nace,
    case when aveg is null then 0 else aveg end as aveg
from (select 
            '00411' as reporting_soc,
            sociedade_contraparte as counterparty_soc,
            cod_ajust as adjustment_code,
            concat(idcomb_satelite,";",excluded_paris,";",env_sust_ccm,";",maturity_esg,";",chronic_changes,";",acute_changes) as id_comb,
            -round(sum(amount),0) as amount,
            geo,
            cnael as cnae,
            nace_esg as nace,
            ceiling(sum(years_to_maturity * aveg_weight)) as aveg,
            maturity_esg
        
        from bu_esg_work.rf_pilar3_sat79_aux5
        group by 1,2,3,4,6,7,8,10
        order by 4,1,2,3,6,7,8 ) a
where amount <> 0
;

-- tabela para atribuição de métricas
drop table bu_captools_work.rf_pilar3_sat79_tabadj1;-----------Tabela dos adjudicados
create table bu_captools_work.rf_pilar3_sat79_tabadj1 as
select 
        a.idcomb_satelite, 
        a.sociedade_contraparte,
        a.cod_ajust,
        a.cargabal_ct,
        a.saldo_ct,
        b.cod_imovel,
        b.tipo_adjudicado,
        b.cargabal_vc,
        c.cargabal_prov,
        coalesce(b.fiabilidad,c.fiabilidad) as fiabilidad,
        coalesce(b.clase_energetica,c.clase_energetica) as clase_energetica,
        coalesce(b.emisiones,c.emisiones) as emisiones,
        coalesce(b.consumos,c.consumos) as consumos,
        'EXC2' as excluded_paris,
			case
            when b.clase_energetica like 'A%' and b.fiabilidad in ('','1-REAL') then 'ENSM1'
            else 'ENSM2'
			end as 
		env_sust_ccm,
			'MESG4' 
		as maturity_esg,
        case
            when coalesce(prhipotec1.rs_acute,prhipotec2.rs_acute) = 'Yes' then 'ACUT1'
            else 'ACUT2'
        end as acute_changes,
        case
            when coalesce(prhipotec1.rs_chronic,prhipotec2.rs_chronic) = 'Yes' then 'CHRO1'
            else 'CHRO2'
        end as chronic_changes,
        case 
            when cargabal_ct = '' then -saldo_ct
            when b.cargabal_vc in ('1605000','1605010') then b.valor_cargabal_vc
            else -c.valor_cargabal_prov
        end as amount
      
from (  select csatelite, idcomb_satelite, sociedade_contraparte, cod_ajust, cargabal_ct, sum(saldo_ct) as saldo_ct 
        from bu_captools_work.rf_pilar3_universo_full where csatelite = 79 and idcomb_satelite like '%MC10%' 
        group by 1,2,3,4,5) as a
        
left join (select * from --Tabela final dos adjudicados depois de voltar da gloval 
            where tipo_adjudicado <> 'Totta URBE' and cargabal_vc in ('1605000','1605010')) as b
on a.cargabal_ct = b.cargabal_vc
        
left join (select * from --Tabela final dos adjudicados depois de voltar da gloval  
            where tipo_adjudicado <> 'Totta URBE' and cargabal_prov in ('2642300')) as c
on a.cargabal_ct = c.cargabal_prov

-- risco físico real
left join (select * from bu_captools_work.pilar3_physical_riskv2 where nace_code like '%Secured%') as prhipotec1
on coalesce(b.id3_collateral_nuts,c.id3_collateral_nuts) = prhipotec1.nuts_code

-- risco físico proxy
left join (select * from bu_captools_work.pilar3_proxy_physical_risk where nace_code like '%Secured%') as prhipotec2
on prhipotec2.iso_code = 620
;
